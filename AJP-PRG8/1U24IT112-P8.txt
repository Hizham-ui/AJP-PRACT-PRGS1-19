public class SelectionSort {

    // Generic method for Selection Sort
    // T must implement Comparable<T>
    public static <T extends Comparable<T>> void selectionSort(T[] array) {
        int n = array.length;

        // Move the boundary of the unsorted subarray
        for (int i = 0; i < n - 1; i++) {

            // Assume the first element is the minimum
            int minIndex = i;

            // Find the minimum element
            for (int j = i + 1; j < n; j++) {
                if (array[j].compareTo(array[minIndex]) < 0) {
                    minIndex = j;
                }
            }

            // Swap the minimum element with the first element
            T temp = array[minIndex];
            array[minIndex] = array[i];
            array[i] = temp;
        }
    }

    // Generic method to print an array
    public static <T> void printArray(T[] array) {
        for (T element : array) {
            System.out.print(element + " ");
        }
        System.out.println();
    }

    public static void main(String[] args) {

        // Integer array
        Integer[] intArray = {5, 3, 9, 1, 7};

        // Float array
        Float[] floatArray = {5.5f, 3.3f, 9.9f, 1.1f, 7.7f};

        System.out.println("Original Integer Array:");
        printArray(intArray);

        System.out.println("Original Float Array:");
        printArray(floatArray);

        // Sort both arrays using the same generic method
        selectionSort(intArray);
        selectionSort(floatArray);

        System.out.println("Sorted Integer Array:");
        printArray(intArray);

        System.out.println("Sorted Float Array:");
        printArray(floatArray);
    }
}
