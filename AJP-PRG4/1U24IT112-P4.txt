import java.time.LocalDate;
import java.time.Period;
import java.time.format.DateTimeFormatter;
import java.util.Scanner;

class Practical4
{
    public static void main(String[] args) {

        Scanner sc = new Scanner(System.in);

        System.out.println("Enter the month you were born (1 to 12):");
        int month = sc.nextInt();

        System.out.println("Enter the day of the month you were born:");
        int day = sc.nextInt();

        System.out.println("Enter the year you were born (four digits):");
        int year = sc.nextInt();

        LocalDate birthDate = LocalDate.of(1997, 07, 05);

        DateTimeFormatter formatter = DateTimeFormatter.ofPattern("MMMM dd, yyyy");
        System.out.println("Your birth date is " + birthDate.format(formatter));

        LocalDate currentDate = LocalDate.now();
        System.out.println("Today's date is " + currentDate.format(formatter));

        Period age = Period.between(birthDate, currentDate);
        System.out.println("Your age is: " + age.getYears() + " years.");

        sc.close();
    }
}
